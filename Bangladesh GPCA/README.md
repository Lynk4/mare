```jsx
RC4 (ARC4)
/*
 * ARC4 key schedule
 */
void mbedtls_arc4_setup( mbedtls_arc4_context *ctx, const unsigned char *key,
				 unsigned int keylen )
{
	int i, j, a;
	unsigned int k;
	unsigned char *m;
	ctx->x = 0;
	ctx->y = 0;
	m = ctx->m;
	for( i = 0; i < 256; i++ )
		m[i] = (unsigned char) i;
	j = k = 0;
	for( i = 0; i < 256; i++, k++ )
	{
		if( k >= keylen ) k = 0;
		a = m[i];
		j = ( j + a + key[k] ) & 0xFF;
		m[i] = m[j];
		m[j] = (unsigned char) a;
	}
}
```
